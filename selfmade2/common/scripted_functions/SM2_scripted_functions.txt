can_decrease_autonomy = {
	condition = {
		tooltip = "AI SHOULD NOT DECREASE AUTONOMY WHEN EFFECT EXCEEDS"
		potential = {
			owner = {
				ai = yes
			}
		}
		allow = {
			local_autonomy_above_min = 10
		}
	}
	## 开局别乱动
	condition = {
		tooltip = "BEFORE 1446"
		potential = { 
			NOT = {
				is_year = 1446
			}
		}
		allow = {
			always = no
		}
	}
	## ai别乱点
	condition = {
		tooltip = "AI: YOU CAN JUST WAIT"
		potential = { 
			owner = {
				ai = yes
			}
			any_neighbor_province = {
				has_province_modifier = autonomy_decreased
			}
		}
		allow = {
			always = no
		}
	}## ai别乱点
	condition = {
		tooltip = "AI: BE CAREFUL, UNREST IS HIGH"
		potential = { 
			owner = {
				ai = yes
			}
			OR = {
				AND = {
					NOT = {
						has_province_modifier = recent_uprising
					}
					unrest = -8
				}
				unrest = -15
			}
		}
		allow = {
			always = no
		}
	}## 先让ai别动了
	condition = {
		tooltip = "AI DONT TOUCH"
		potential = { 
			owner = {
				ai = yes
			}
		}
		allow = {
			always = no
		}
	}
}
