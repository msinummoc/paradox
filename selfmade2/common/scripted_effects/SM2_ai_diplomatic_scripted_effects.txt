SM2_historical_war_target_effects = {
	if = {
		limit = {
			tag = TUR
		}
		every_neighbor_country = {
			limit = {
				NOT = {
					has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_selected
					}
				}
				OR = {
					any_owned_province = {
						OR = {
							is_core = ROOT
							is_permanent_claim = ROOT
						}
						OR = {
							region = anatolia_region
							region = balkan_region
						}
					}
					tag = BYZ
				}
				
			}
			if = {
				limit = { NOT = { alliance_with = ROOT } }
				add_opinion = { who = ROOT modifier = SM2_war_target_t1 }
			}
			SM2_record_war_target_with_ally_check = yes
		
		}

	}else_if = {
		limit = {
			government_rank = 2
		}
		capital = {
			region = {
				save_event_target_as = SM2_capital_region
			}
		}
		every_neighbor_country = {
			limit = {
				NOT = {
					has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_selected
					}
				}
				OR = {
					any_owned_province = {
						region = event_target:SM2_capital_region
						OR = {
							is_core = ROOT
							is_permanent_claim = ROOT
						}
					}
					
				}
			}
			if = {
				limit = { NOT = { alliance_with = ROOT } }
				add_opinion = { who = ROOT modifier = SM2_war_target_t1 }
			}
			SM2_record_war_target_with_ally_check = yes
		}
	}

}

# 记录目标（含盟友检查）：盟友则互减信任，不计入选择；非盟友正常记录


SM2_record_war_target_with_ally_check = {
	if = {
		limit = { alliance_with = ROOT }
		# 盟友：减信任，不计入目标
		add_trust = {
			who = ROOT
			value = -10
			mutual = yes
		}
	}
	# 不要改，else会导致add_opinion不生效
	if = {
		limit = { NOT = { alliance_with = ROOT } }
		# 非盟友：添加标记并更新计数器
		add_opinion = {
			who = ROOT
			modifier = SM2_declare_war_selected
		}
		ROOT = {
			# 增加实际选中计数
			change_variable = {
				which = SM2_selected_targets_count
				value = 1
			}
			# 减少剩余需求
			change_variable = {
				which = SM2_need_select_count
				value = -1
			}
		}
	}
}
