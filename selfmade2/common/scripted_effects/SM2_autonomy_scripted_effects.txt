SM2_autonomy_init_effect = {
	#自治度为100时，自治度自然降低的值（100意味着100自治度时自治度每月-1.0）
	set_variable = {
		which = SM2_autonomy_decade
		value = 100
	}
	#自治度增加基础值
	#如果数值为1，意味着会趋近于邻居省份的最低值（不考虑自身自治度降低导致自治度增加值提高）
	set_variable = {
		which = SM2_autonomy_base
		value = 0.85
	}

	set_variable = {
		which = SM2_autonomy_upper
		value = 0.9
	}

	set_variable = {
		which = SM2_old_autonomy
		value = 100
	}
	# set_variable = {
	# 	which = SM2_autonomy_different_country
	# 	value = -0.25
	# }
	#地形修正直接到下面去改
	# #趋近速度为100，则（按照当前恒定速度）1个月就会完成趋近
	# set_variable = {
	# 	which = SM2_autonomy_speed_base
	# 	value = 2
	# }

	## 根据趋近速度和自治度自然增长的速度，以及最终希望趋近于的数值，可以得到一个实际的比值

	#每点叛乱增加的自治度
	set_variable = {
		which = SM2_autonomy_per_unrest
		value = 0.02
	}
	
	# 缓存地形与气候修正（不可变），避免每月重复判断
	set_variable = {
		which = SM2_autonomy_terrain_modifier
		value = 0
	}
	if = {
		limit = { has_terrain = glacier }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.4 }
	}
	else_if = {
		limit = { has_terrain = farmlands }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.05 }
	}
	else_if = {
		limit = { has_terrain = forest }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.15 }
	}
	else_if = {
		limit = { has_terrain = hills }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.2 }
	}
	else_if = {
		limit = { has_terrain = woods }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.15 }
	}
	else_if = {
		limit = { has_terrain = mountain }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.3 }
	}
	else_if = {
		limit = { has_terrain = grasslands }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.1 }
	}
	else_if = {
		limit = { has_terrain = jungle }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.3 }
	}
	else_if = {
		limit = { has_terrain = marsh }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.25 }
	}
	else_if = {
		limit = { has_terrain = desert }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.2 }
	}
	else_if = {
		limit = { has_terrain = coastal_desert }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.15 }
	}
	else_if = {
		limit = { has_terrain = coastline }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.1 }
	}
	else_if = {
		limit = { has_terrain = drylands }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.1 }
	}
	else_if = {
		limit = { has_terrain = highlands }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.2 }
	}
	else_if = {
		limit = { has_terrain = savannah }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.2 }
	}
	else_if = {
		limit = { has_terrain = steppe }
		set_variable = { which = SM2_autonomy_terrain_modifier value = 0.2 }
	}
	
	if = {
		limit = { has_climate = tropical }
		change_variable = { which = SM2_autonomy_terrain_modifier value = 0.05 }
	}
	else_if = {
		limit = { has_climate = arctic }
		change_variable = { which = SM2_autonomy_terrain_modifier value = 0.05 }
	}
	
	# 为省份随机分配4个组，用于分批计算
	random_list = {
		1 = { set_province_flag = SM2_autonomy_group_a }
		1 = { set_province_flag = SM2_autonomy_group_b }
		1 = { set_province_flag = SM2_autonomy_group_c }
		1 = { set_province_flag = SM2_autonomy_group_d }
	}
}

# 用来在最开始的一年计算出所有省份应有的自治度数值
SM2_autonomy_at_start_effect = {
	#本地自治度 注意是trigger_value，否则会得出变化量
	export_to_variable = {
		which = var1
		value = trigger_value:local_autonomy
	}
	set_variable = {
		which = var2
		which = var1
	}
	#获得周围自治度的最低值
	every_neighbor_province = {
		limit = {
			owned_by = ROOT
		}
		export_to_variable = {
			which = var2
			value = trigger_value:local_autonomy
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = var2
						which = PREV
					}
				}
			}
			PREV = {
				set_variable = {
					which = var2
					which = PREV
				}
			}
		}
	}
	# 同一海区的也算
	if = {
		limit = {
			has_port = yes
		}
		sea_zone = {
			set_variable = {
				which = var2
				which = PREV
			}
			every_neighbor_province = {
				limit = {
					owned_by = ROOT
				}
				export_to_variable = {
					which = var2

				}

								which = var2

							}

					PREV = {
						set_variable = {
							which = var2
							which = PREV
						}
					}
				}
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = var2
							which = PREV
						}
					}
				}
				PREV = {
					set_variable = {
						which = var2
						which = PREV
					}
				}
			}
		}
	}
	change_variable = {
		which = var2
		value = 0.5
	}
	#加一下避免自己和自己比大小还赢了
	if = {
		limit = {
			check_variable = {
				which = var1
				which = var2
			}
		}
		subtract_variable = {
			which = var1
			which = var2
		}
		

		SM2_autonomy_modifier_effect = yes

		#这里直接按照比例加减就可以了，但是注意，比值只是在当前自治度均衡的情况下成立。也就是说需要根据local_autonomy的变化值得到一个均衡值计算，而不是直接用当前的自治度计算

		export_to_variable = {
			which = var3
			value = modifier:local_autonomy
		}
		export_to_variable = {
			which = var4
			value = modifier:global_autonomy
			who = owner
		}
		change_variable = {
			which = var3
			which = var4
		}
		multiply_variable = {
			which = var3
			value = 10000
		}
		divide_variable = {
			which = var3
			which = SM2_autonomy_decade
		}

		change_variable = {
			which = var1
			which = var3
		}
		multiply_variable = {
			which = var1
			which = var2
		}
		subtract_variable = {
			which = var1
			which = var3
		}

		if = {
			limit = {
				check_variable = {
					which = var1
					value = 64
				}
			}
			add_local_autonomy = -64
			subtract_variable = {
				which = var1
				value = 64
			}
		}
		if = {
			limit = {
				check_variable = {
					which = var1
					value = 32
				}
			}
			add_local_autonomy = -32
			subtract_variable = {
				which = var1
				value = 32
			}
		}
		if = {
			limit = {
				check_variable = {
					which = var1
					value = 16
				}
			}
			add_local_autonomy = -16
			subtract_variable = {
				which = var1
				value = 16
			}
		}
		if = {
			limit = {
				check_variable = {
					which = var1
					value = 8
				}
			}
			add_local_autonomy = -8
			subtract_variable = {
				which = var1
				value = 8
			}
		}
		if = {
			limit = {
				check_variable = {
					which = var1
					value = 4
				}
			}
			add_local_autonomy = -4
			subtract_variable = {
				which = var1

	}
	export_to_variable = {
		which = var2
		value = modifier:global_autonomy
		who = owner
	}
	change_variable = {
		which = var1
		which = var2
	}
	multiply_variable = {
		which = var1
		value = 10000
	}
	divide_variable = {
		which = var1
		which = SM2_autonomy_decade
	}
	#随机修正
	subtract_variable = {
		which = var1
		value = 5
	}
	export_to_variable = {
		which = var3
		value = modifier:min_local_autonomy
	}
	set_variable = {
		which = var4
		value = 0
	}

	if = {
		limit = {

		}
		clr_province_flag = SM2_autonomy_recalculate
		set_province_flag = SM2_autonomy_initialized
		
		# 初始化广播变量（neighbor_min已经在old中设置）
		export_to_variable = {
			which = SM2_autonomy_last_broadcast
			value = trigger_value:local_autonomy
		}
	}
	else = {
		# 正常的广播机制
		
		# 1. 获取当前自治度
		export_to_variable = {
			which = var1
			value = trigger_value:local_autonomy
		}
		
		# 2. 初始化机制
		if = {
			limit = {
				NOT = { has_province_flag = SM2_autonomy_initialized }
			}
			# 标记初始化状态
			set_province_flag = SM2_autonomy_initialized
			
			# 暂存当前值
			set_variable = {
				which = SM2_autonomy_last_broadcast
				which = var1
			}
			set_variable = {
				which = SM2_autonomy_neighbor_min
				which = var1
			}
			
			# 标记下个月重新计算以获取正确值
			set_province_flag = SM2_autonomy_recalculate
		}
		
		# 3. 检查是否需要广播（差值>=3）
		set_variable = {
			which = var2
			which = var1
		}
		subtract_variable = {
			which = var2
			which = SM2_autonomy_last_broadcast
		}
		
		# 取绝对值
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = var2
						value = 0
					}
				}
			}
			multiply_variable = {
				which = var2
				value = -1
			}
		}
		
		# 4. 如果差值>=3，广播给周围省份
		if = {
			limit = {
				check_variable = {
					which = var2

				}
			}
			
			# 更新Last Broadcast
			set_variable = {
				which = SM2_autonomy_last_broadcast
				which = var1
			}
			
			# 准备广播数据
			set_variable = {
				which = SM2_broadcast_value
				which = var1
			}
			
			# 广播给陆地邻居
			every_neighbor_province = {
				limit = {
					owned_by = ROOT
				}
				
				# 获取广播值
				set_variable = {
					which = SM2_broadcast_value
					which = PREV
				}
				
				# 如果广播值 < 邻居保存的最低值，则更新邻居的最低值
				# (只有变小才更新，忽略变大)
				if = {
					limit = {
						NOT = {
							check_variable = {
								which = SM2_autonomy_neighbor_min
								which = SM2_broadcast_value
							}
						}
					}
					set_variable = {
						which = SM2_autonomy_neighbor_min
						which = SM2_broadcast_value
					}
				}
			}
			
			# 广播给海区邻居
			if = {
				limit = {
					has_port = yes
				}
				sea_zone = {
					# 传给海区
					set_variable = {
						which = SM2_broadcast_value
						which = PREV
					}
					every_neighbor_province = {
						limit = {
							owned_by = ROOT
							has_port = yes
						}
						
						# 从海区获取
						set_variable = {
							which = SM2_broadcast_value
							which = PREV
						}
						
						# 如果广播值 < 邻居保存的最低值，则更新邻居的最低值
						if = {
							limit = {
								NOT = {
									check_variable = {
										which = SM2_autonomy_neighbor_min
										which = SM2_broadcast_value
									}
								}
							}
							set_variable = {
								which = SM2_autonomy_neighbor_min
								which = SM2_broadcast_value
							}
						}
					}
				}
			}
		}
		
		# 5. 根据保存的neighbor_min计算变化量
		set_variable = {
			which = var3
			which = var1
		}
		
		# 自增机制：防止“坏消息”缺失导致数值永远偏低
