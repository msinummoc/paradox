SM2_autonomy_init_effect = {
	#自治度为100时，自治度自然降低的值（100意味着100自治度时自治度每月-1.0）
	set_variable = {
		which = SM2_autonomy_decade
		value = 100
	}
	#自治度增加基础值
	#如果数值为1，意味着会趋近于邻居省份的最低值
	set_variable = {
		which = SM2_autonomy_base
		value = 0.85
	}
	# set_variable = {
	# 	which = SM2_autonomy_different_country
	# 	value = -0.25
	# }
	#地形修正直接到下面去改
	#趋近速度为100，则1个月就会完成趋近
	set_variable = {
		which = SM2_autonomy_speed_base
		value = 1
	}
}


SM2_autonomy_effect = {

		which = var2
		which = var1
	}
	#获得周围自治度的最低值
	every_neighbor_province = {
		limit = {
			owned_by = ROOT
		}

			which = var2

		if = {
			limit = {

						which = var2
						which = PREV
					}

					which = var2

				}
			}
		}
	}

		which = var2
		value = 0.5
	}

SM2_autonomy_game_start_effect = {
	export_to_variable = {
		which = var1
		value = modifier:local_autonomy
	}
	export_to_variable = {
		which = var2
		value = modifier:global_autonomy
		who = owner
	}
	change_variable = {
		which = var1
		which = var2
	}
	multiply_variable = {
		which = var1
		value = 10000
	}
	divide_variable = {
		which = var1
		which = SM2_autonomy_decade
	}
	#修正
	subtract_variable = {
		which = var1
		value = 5
	}
	export_to_variable = {
		which = var3
		value = modifier:min_local_autonomy
	}
	set_variable = {
		which = var4
		value = 0
	}
	if = {
		limit = {
			local_autonomy_above_min = 64
		}
		add_local_autonomy = -64
		change_variable = {
			which = var4
			value = 64
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 32
		}
		add_local_autonomy = -32
		change_variable = {
			which = var4
			value = 32
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 16
		}
		add_local_autonomy = -16
		change_variable = {
			which = var4
			value = 16
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 8
		}
		add_local_autonomy = -8
		change_variable = {
			which = var4
			value = 8
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 4
		}
		add_local_autonomy = -4
		change_variable = {
			which = var4
			value = 4
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 2
		}
		add_local_autonomy = -2
		change_variable = {
			which = var4
			value = 2
		}
	}
	if = {
		limit = {
			local_autonomy_above_min = 1
		}
		add_local_autonomy = -1
		change_variable = {
			which = var4
			value = 1
		}
	}

	if = {
		limit = {
			check_variable = {
				which = var4
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = var1
						value = 0
					}
				}
			}
			set_variable = {
				which = var1
				which = var4
			}
		}else = {
			change_variable = {
				which = var1
				which = var4
			}
		}
	}else = {
		add_local_autonomy = -100
		change_variable = {
			which = var4
			which = var1
		}
		if = {
			limit = {
				check_variable = {
					which = var4
					which = var3
				}
			}
			set_variable = {
				which = var1
				which = var4
			}
		}else = {
			set_variable = {
				which = var1
				which = var3
			}
		}
	}

	#conclude
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 64
			}
		}
		add_local_autonomy = 64
		subtract_variable = {
			which = var1
			value = 64
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 32
			}
		}
		add_local_autonomy = 32
		subtract_variable = {
			which = var1
			value = 32
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 16
			}
		}
		add_local_autonomy = 16
		subtract_variable = {
			which = var1
			value = 16
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 8
			}
		}
		add_local_autonomy = 8
		subtract_variable = {
			which = var1
			value = 8
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 4
			}
		}
		add_local_autonomy = 4
		subtract_variable = {
			which = var1
			value = 4
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 2
			}
		}
		add_local_autonomy = 2
		subtract_variable = {
			which = var1
			value = 2
		}
	}
	if = {
		limit = {
			check_variable = {
				which = var1
				value = 1
			}
		}
		add_local_autonomy = 1
		subtract_variable = {
			which = var1
			value = 1
		}
	}

}
