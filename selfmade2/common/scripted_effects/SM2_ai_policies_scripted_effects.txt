SM_ai_policies_yearly_effect = {
	if = {
		limit = { NOT = { has_country_flag = SM2_policy_initialized } }
		SM2_initialize_policies_by_region = yes
	}
	random_list = {
		40 = {}
		25 = {
			if = {
				limit = {
					OR = {
						is_at_war = yes
						#has_country_modifier = SM2_declare_war_active
						#has_country_modifier = SM2_declare_war_chosen_in_three_years
					}
				}
				SM2_policy_cycle_land_offense = yes
			}
		}
		25 = {
			if = {
				limit = {
					OR = {
						is_at_war = yes
						#has_country_modifier = SM2_declare_war_active
					}
				}
				SM2_policy_cycle_land_defense = yes
			}
		}
		20 = {
			if = {
				limit = { navy_size_percentage = 0.6 }
				SM2_policy_cycle_navy_infrastructure = yes
			}
			else = {
				SM2_policy_cycle_army_infrastructure = yes
			}
		}
		20 = {
			if = {
				limit = { war_exhaustion = 2 }
				SM2_policy_cycle_reform = yes
			}
			else = {
				SM2_policy_cycle_centralization = yes
			}
		}
		20 = {
			SM2_policy_cycle_force_quality = yes
		}
		15 = {
			SM2_policy_cycle_diplomacy = yes
		}
		15 = {
			SM2_policy_cycle_trade = yes
		}
		10 = {
			SM2_policy_cycle_capital_focus = yes
		}
	}
}
