	potential = {
		any_country = {
			technology_group = ROOT
			tech_difference = 1
		}
		NOT = {
			any_country = {
				technology_group = ROOT
				tech_difference = 2
			}
		}
	}

		any_country = {
			technology_group = ROOT
			tech_difference = 1
		}
	}
	technology_cost = -0.05
	global_institution_spread = 0.05
}

SM2_national_defense = {
	potential = {
		is_at_war = yes
	}
	trigger = {
		OR = {
			is_in_war = {
				attacker_leader = ROOT
			}
			is_in_war = {
				defender_leader = ROOT
			}
			AND = {
				OR = {
					NOT = {
						war_score = -15
					}
					capital_scope = {
						has_siege = yes
					}
				}
				is_at_war = yes
			}
			num_of_rebel_controlled_provinces = 1
		}
	}
	land_maintenance_modifier = -0.2
	naval_maintenance_modifier = -0.2
	manpower_recovery_speed = 1
	war_exhaustion = 0.1
	defensiveness = 1
	land_forcelimit_modifier = 0.5
}

SM2_not_at_war = {
	potential = {
		is_at_war = no
	}
	trigger = {
		NOT = {
			num_of_rebel_controlled_provinces = 1
		}
		NOT = {
			any_country = {
				OR = {
					has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_prepared
					}
					has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_active
					}
					reverse_has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_prepared
					}
					reverse_has_opinion_modifier = {
						who = ROOT
						modifier = SM2_declare_war_active
					}
				}
			}
		}
	}
	land_morale = -0.1
	discipline = -0.1
	naval_morale = -0.1
}
