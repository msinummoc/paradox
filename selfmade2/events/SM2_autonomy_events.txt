namespace = SM2_autonomy



#和叛乱度一样，自治度也会传播，也就是低自治度的省份会降低周围省份的自治度。跨国的情况也成立，系数自然更低

#游戏开始后执行，每30天计算一次自治度
country_event = {
	id = SM2_autonomy.1
	title = SM2_autonomy.1.t
	desc = SM2_autonomy.1.d
	picture = INVENTION_eventPicture
	hidden = yes
	is_triggered_only = yes 
	trigger = {
		always = yes
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_country_flag = SM2_autonomy_init_flag
				}
			}
			every_owned_province = {
				SM2_autonomy_init_effect = yes
				SM2_autonomy_game_start_effect = yes
			}
			set_country_flag = SM2_autonomy_init_flag
		}
		every_owned_province = {
			SM2_autonomy_effect = yes
		}
		country_event = { 
			id = SM2_autonomy.1
			days = 30
		}
    }
	
	option = {
		
	}
}
