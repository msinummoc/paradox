namespace = SM2_unrest_spread


#游戏开始后执行，每30天计算一次叛乱度
country_event = {
	id = SM2_unrest_spread.1
	title = SM2_unrest_spread.1.t
	desc = SM2_unrest_spread.1.d
	picture = INVENTION_eventPicture
	hidden = yes
	is_triggered_only = yes 
	trigger = {
		always = yes
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_country_flag = SM2_unrest_spread_init_flag
				}
			}
			every_owned_province = {
				SM2_unrest_spread_province_init_effect = yes
			}
			set_country_flag = SM2_unrest_spread_init_flag
		}
		every_owned_province = {
			SM2_unrest_spread_province_effect = yes
		}
		country_event = { 
			id = SM2_unrest_spread.1
			days = 30
		}
    }
	
	option = {
		
	}
}
