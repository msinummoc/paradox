namespace = SM2_unrest_spread


#游戏开始后执行
country_event = {
	id = SM2_unrest_spread.1
	title = SM2_unrest_spread.1.t
	desc = SM2_unrest_spread.1.d
	picture = INVENTION_eventPicture
	hidden = yes
	is_triggered_only = yes
	trigger = {
		always = yes
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_country_flag = SM2_unrest_spread_init_flag
				}
			}
			every_owned_province = {
				SM2_unrest_spread_province_init_effect = yes
			}
			set_country_flag = SM2_unrest_spread_init_flag
		}
		
    }
	
	option = {
		
	}
}

# 每30天计算一次叛乱度
country_event = {
	id = SM2_unrest_spread.2
	title = SM2_unrest_spread.2.t
	desc = SM2_unrest_spread.2.d
	picture = INVENTION_eventPicture
	hidden = yes
	is_triggered_only = yes
	trigger = {
		always = yes
	}
	immediate = {
		# 如果省份数>40，分成4个子事件处理；否则直接处理
		if = {
			limit = {
				num_of_provinces_in_states = 150
			}
			country_event = { id = SM2_unrest_spread.20 }
			country_event = { id = SM2_unrest_spread.21 }
			country_event = { id = SM2_unrest_spread.22 }
			country_event = { id = SM2_unrest_spread.23 }
		}
		else = {
			every_owned_province = {
				SM2_unrest_spread_province_effect = yes
			}
		}
    }
	
	option = {
		
	}
}
