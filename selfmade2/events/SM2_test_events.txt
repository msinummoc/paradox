country_event = {
	id = SM2_test.2
	title = SM2_test.2.t
	desc = SM2_test.2.d
	picture = INVENTION_eventPicture
	hidden = no
	is_triggered_only = yes
	trigger = {
		tag = MNG
	}
	immediate = {
		set_variable = { which = test_start_time value = 1 }
		every_owned_province = {
			SM2_test_heavy_computation_effect = yes
		}
		set_variable = { which = test_end_time value = 1 }
	}
	
	option = {
		name = SM2_test.2.a
		custom_tooltip = SM2_test.2.a.tt
	}
}

# 测试方法B：并行处理（分组+多个country_event）
country_event = {
	id = SM2_test.20
	title = SM2_test.20.t
	desc = SM2_test.20.d
	picture = INVENTION_eventPicture
	hidden = no
	is_triggered_only = yes
	trigger = {
		tag = MNG
	}
	immediate = {
		set_variable = { which = test_start_time value = 1 }
		# 触发8个事件分别处理各组
		country_event = { id = SM2_test.21 }
		country_event = { id = SM2_test.22 }
		country_event = { id = SM2_test.23 }
		country_event = { id = SM2_test.24 }
		country_event = { id = SM2_test.25 }
		country_event = { id = SM2_test.26 }
		country_event = { id = SM2_test.27 }
		country_event = { id = SM2_test.28 }
		set_variable = { which = test_end_time value = 1 }
	}
	
	option = {
		name = SM2_test.20.a
		custom_tooltip = SM2_test.20.a.tt
	}
}
